void generateSalesSummary(IEnumerable<string> salesFiles, double salesTotal)
{
    string salesSummary = "Sales Sumary\n\tTotal Sales: $" + salesTotal + "\n\n\tDetails:";

    foreach (var file in salesFiles)
    {
        if (file.EndsWith("salestotals.json"))
        {
            string salesJson = File.ReadAllText(file);
            SalesTotalsData? data = JsonConvert.DeserializeObject<SalesTotalsData?>(salesJson);
            var salesAmount = data?.OverallTotal ?? 0;

            salesSummary += "\n\t\t" + file + ": $" + salesAmount;
        }
    }

    File.AppendAllText(Path.Combine(salesTotalDir, "salesSummary.txt"), $"{salesSummary}{Environment.NewLine}");
}
